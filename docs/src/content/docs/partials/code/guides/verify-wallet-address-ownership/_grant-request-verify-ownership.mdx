if err := requestBody.FromGrantRequestWithSubject(as.GrantRequestWithSubject{
  Subject: as.Subject{
    SubIds: []struct {
      Format as.SubjectSubIdsFormat `json:"format"`
      Id     string                 `json:"id"`
    }{
      {Format: as.Uri, Id: *userWalletAddress.Id},
    },
  },
  Interact: as.InteractRequest{
    Start: []as.InteractRequestStart{as.InteractRequestStartRedirect},
    Finish: &struct {
      Method as.InteractRequestFinishMethod `json:"method"`
      Nonce  string                         `json:"nonce"`
      Uri    string                         `json:"uri"`
    }{
      Method: as.InteractRequestFinishMethodRedirect,
      Uri:    "https://paymentplatform.example/finish/{...}", // where to redirect the user to after they've completed the interaction

    },

}); err != nil {
